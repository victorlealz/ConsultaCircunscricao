        delegaciasData.forEach(del => {
            del.bairros.forEach(bairro => {
                bairroMap[bairro.toLowerCase()] = {
                    delegacia: del.id,
                    endereco: del.endereco,

                    bairroOficial: bairro,

        const sinonimos = {
            "jacarapé": "Costa do Sol",
            "colinas do sul": "Gramame"
        };

        const searchInput = document.getElementById('search-input');
        const suggestionsBox = document.getElementById('suggestions');
        const resultCard = document.getElementById('result-card');
        const resDelegacia = document.getElementById('res-delegacia');
        const resBairro = document.getElementById('res-bairro');
        const resEndereco = document.getElementById('res-endereco');
        const resTelefone = document.getElementById('res-telefone');
        const resObs = document.getElementById('res-obs');
        const resMapsBtn = document.getElementById('res-maps-btn');

        const allTerms = [...Object.keys(bairroMap), ...Object.keys(sinonimos)];

        searchInput.addEventListener('input', function() {
            const query = this.value.trim().toLowerCase();
            suggestionsBox.innerHTML = '';
            
            if (query.length === 0) {
                suggestionsBox.style.display = 'none';
                resultCard.style.display = 'none';
                return;
            }

            const matches = allTerms.filter(term => term.includes(query));

            if (matches.length > 0) {
                suggestionsBox.style.display = 'block';
                matches.forEach(match => {
                    const item = document.createElement('div');
                    item.className = 'suggestion-item';
                    let display = match;
                    if (sinonimos[match]) {
                        display = `${match} (Localidade de ${sinonimos[match]})`;
                    }
                    item.textContent = display.charAt(0).toUpperCase() + display.slice(1);
                    item.addEventListener('click', function() {
                        searchInput.value = match.charAt(0).toUpperCase() + match.slice(1);
                        suggestionsBox.style.display = 'none';
                        executarConsulta(match);
                    });
                    suggestionsBox.appendChild(item);
                });
            } else {
                suggestionsBox.style.display = 'none';
            }

            if (bairroMap[query]) {
                executarConsulta(query);
            } else if (sinonimos[query]) {
                executarConsulta(sinonimos[query].toLowerCase());
            } else {
                resultCard.style.display = 'none';
            }
        });

        function executarConsulta(termo) {
            let target = termo;
            if (sinonimos[termo]) {
                target = sinonimos[termo].toLowerCase();
            }
            const info = bairroMap[target];
            if (info) {
                resDelegacia.textContent = info.delegacia;
                resBairro.textContent = info.bairroOficial;
                resEndereco.textContent = info.endereco;
                resTelefone.textContent = info.telefone;

                if (info.obs) {
                    resObs.textContent = info.obs;
                    resObs.style.display = 'block';
                } else {
                    resObs.style.display = 'none';
                }

                // Gera a URL do Google Maps com o nome da delegacia e o endereço
                const searchQuery = encodeURIComponent(`${info.delegacia} ${info.endereco}`);
                resMapsBtn.href = `https://www.google.com/maps/search/?api=1&query=${searchQuery}`;

                resultCard.style.display = 'block';
            }
        }

        document.addEventListener('click', function(e) {
            if (!e.target.closest('.search-container')) {
                suggestionsBox.style.display = 'none';
            }
        });
